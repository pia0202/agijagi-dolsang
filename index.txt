        function formatDateWithDay(dateObj) {
            const year = dateObj.getFullYear();
            const month = String(dateObj.getMonth() + 1).padStart(2, '0');
            const day = String(dateObj.getDate()).padStart(2, '0');
            const dayOfWeek = DAYS_OF_WEEK[dateObj.getDay()];
            return `${day}/${month}/${year} (${dayOfWeek})`;
        }

        function formatIsoToCustomDate(isoStr) {
            if (!isoStr) return '미선택';
            const parts = isoStr.split('-');
            if (parts.length !== 3) return isoStr;
            const year = parts[0];
            const month = parts[1];
            const day = parts[2];
            const d = new Date(parseInt(year), parseInt(month) - 1, parseInt(day));
            const dayOfWeek = DAYS_OF_WEEK[d.getDay()];
            return `${day}/${month}/${year} (${dayOfWeek})`;
        }

        function getIsoFromFormatted(formattedStr) {
            if (!formattedStr) return '';
            if (formattedStr.includes('/')) {
                const parts = formattedStr.split(' ')[0].split('/');
                if (parts.length === 3) {
                    return `${parts[2]}-${parts[1]}-${parts[0]}`;
                }
            }
            return formattedStr.split(' ')[0];
        }

        function getBookedDatesSet() {
            const bookedSet = new Set();
            reservations.forEach(r => {
                if (r.status !== '취소') {
                    const iso = getIsoFromFormatted(r.eventDate);
                    if (iso) bookedSet.add(iso);
                }
            });
            return bookedSet;
        }

        window.renderCalendar = function() {
            const grid = document.getElementById('calendarDaysGrid');
            const titleElem = document.getElementById('calMonthTitle');
            if (!grid || !titleElem) return;

            titleElem.innerText = `${currentCalYear}년 ${currentCalMonth + 1}월`;
            grid.innerHTML = '';

            const firstDayIndex = new Date(currentCalYear, currentCalMonth, 1).getDay();
            const totalDaysInMonth = new Date(currentCalYear, currentCalMonth + 1, 0).getDate();
            const bookedSet = getBookedDatesSet();

            const selectedDateVal = document.getElementById('eventDate').value;
            const todayStrIso = new Date().toISOString().split('T')[0];

            for (let i = 0; i < firstDayIndex; i++) {
                const emptyCell = document.createElement('div');
                emptyCell.className = 'cal-day-cell text-transparent pointer-events-none opacity-0';
                grid.appendChild(emptyCell);
            }

            for (let day = 1; day <= totalDaysInMonth; day++) {
                const dayString = String(day).padStart(2, '0');
                const monthString = String(currentCalMonth + 1).padStart(2, '0');
                const isoStr = `${currentCalYear}-${monthString}-${dayString}`;

                const cell = document.createElement('div');
                const isBooked = bookedSet.has(isoStr);
                const isSelected = selectedDateVal === isoStr;
                const isPast = isoStr < todayStrIso;

                if (isBooked || isPast) {
                    cell.className = 'cal-day-cell cal-day-booked';
                    cell.innerHTML = `<span>${day}</span><span class="text-[8px] sm:text-[9px] ${isBooked ? 'text-indigo-900 font-bold' : 'text-gray-400'}">${isBooked ? '마감' : '지난날'}</span>`;
                    if (isBooked) {
                        cell.onclick = () => showToast('🔒 이미 다른 고객님이 예약 확정한 날짜입니다.');
                    }
                } else if (isSelected) {
                    cell.className = 'cal-day-cell cal-day-selected';
                    cell.innerHTML = `<span>${day}</span><span class="text-[8px] sm:text-[9px] font-bold">선택</span>`;
                    cell.onclick = () => selectCalendarDate(isoStr);
                } else {
                    cell.className = 'cal-day-cell cal-day-available';
                    cell.innerHTML = `<span>${day}</span><span class="text-[8px] sm:text-[9px] text-coral-600 font-normal">가능</span>`;
                    cell.onclick = () => selectCalendarDate(isoStr);
                }

                grid.appendChild(cell);
            }
        };

        window.changeCalMonth = function(offset) {
            currentCalMonth += offset;
            if (currentCalMonth > 11) {
                currentCalMonth = 0;
                currentCalYear += 1;
            } else if (currentCalMonth < 0) {
                currentCalMonth = 11;
                currentCalYear -= 1;
            }
            renderCalendar();
        };

            const bookedSet = getBookedDatesSet();

                showToast('⚠️ 지난 날짜는 선택하실 수 없습니다.');
                return;

        window.searchSingaporeAddress = async function() {
            const pc = document.getElementById('postalCode').value.trim();
            const statusElem = document.getElementById('addressStatus');
            const baseAddrElem = document.getElementById('baseAddress');

            if (pc.length !== 6) {
                statusElem.innerText = '❌ 우편번호 6자리를 정확히 입력해 주세요.';
                statusElem.className = 'text-[10px] sm:text-xs text-coral-600 font-semibold';
                return;
            }

            statusElem.innerText = '🔍 싱가포르 주소 검색 중...';
            statusElem.className = 'text-[10px] sm:text-xs text-coral-700 font-semibold';

            try {
                const response = await fetch(`https://www.onemap.gov.sg/api/common/elastic/search?searchVal=${pc}&returnGeom=N&getAddrDetails=Y`);
                const data = await response.json();

                if (data && data.results && data.results.length > 0) {
                    const result = data.results[0];
                    const roadName = result.ROAD_NAME || '';
                    const building = (result.BUILDING && result.BUILDING !== 'N/A') ? result.BUILDING : '';
                    const blkNo = (result.BLK_NO && result.BLK_NO !== 'N/A') ? `Blk ${result.BLK_NO}` : '';

                    let formattedAddr = [blkNo, roadName, building].filter(Boolean).join(' ');
                    formattedAddr += `, Singapore ${pc}`;
                    baseAddrElem.value = formattedAddr;
                    statusElem.innerText = '✅ 주소가 검색되었습니다.';
                    statusElem.className = 'text-[10px] sm:text-xs text-emerald-600 font-semibold';
                } else {
                    statusElem.innerText = '⚠️ 주소를 찾을 수 없습니다. 직접 입력해 주세요.';
                    statusElem.className = 'text-[10px] sm:text-xs text-amber-600 font-semibold';
                }
            } catch (err) {
                statusElem.innerText = '⚠️ 자동 조회 실패. 직접 주소를 입력해 주세요.';
                statusElem.className = 'text-[10px] sm:text-xs text-red-500 font-semibold';
            }
            updateSummaryPreview();
        };

        function copyToClipboard(str) {
            const el = document.createElement('textarea');
            el.value = str;
            el.setAttribute('readonly', '');
            el.style.position = 'absolute';
            el.style.left = '-9999px';
            document.body.appendChild(el);
            el.select();
            document.execCommand('copy');
            document.body.removeChild(el);
        }

        window.handleFormSubmit = async function(e) {
            e.preventDefault();

            const submitBtn = document.getElementById('submitBtn');
            const rawEventDate = document.getElementById('eventDate').value;
            const eventTime = document.getElementById('eventTime').value;

            if (!rawEventDate) {
                showToast('⚠️ 행사 날짜를 선택해 주세요.');
                return;
            }

            const bookedSet = getBookedDatesSet();
            if (bookedSet.has(rawEventDate)) {
                showToast('🔒 죄송합니다. 이미 다른 고객님이 예약 완료한 날짜입니다.');
                renderCalendar();
                return;
            }

            submitBtn.disabled = true;
            submitBtn.innerHTML = `<span>접수 중...</span>`;

            const formattedEventDate = `${formatIsoToCustomDate(rawEventDate)} ${eventTime}`.trim();

            const baseAddr = document.getElementById('baseAddress').value || '';
            const detailAddr = document.getElementById('detailAddress').value || '';
            const fullLocation = (baseAddr + ' ' + detailAddr).trim();

            const dolTable = document.getElementById('dolTable').value;
            const newDocId = 'AJ-' + Date.now().toString().slice(-6);

            const newReservation = {
                id: newDocId,
                createdDate: document.getElementById('createdDateDisplay').value,
                eventDate: formattedEventDate,
                eventTime: eventTime,
                postalCode: document.getElementById('postalCode').value,
                location: fullLocation,
                customerName: document.getElementById('customerName').value,
                phoneNumber: document.getElementById('phoneNumber').value,
                babyName: document.getElementById('babyName').value,
                babyGender: document.getElementById('babyGender').value,
                dolTable: dolTable,
                foldingScreen: dolTable !== '베이식' ? document.getElementById('foldingScreen').value : '해당없음',
                banner: dolTable === '베이식' ? document.getElementById('banner').value : '해당없음',
                flowers: document.getElementById('flowers').value,
                riceCake: document.getElementById('riceCake').value,
                hanbokMom: toggleStates.hanbokMom ? '신청 (+SGD 250)' : '미신청',
                hanbokDad: toggleStates.hanbokDad ? '신청 (+SGD 250)' : '미신청',
                photo: toggleStates.photo ? '신청 (+SGD 600)' : '미신청',
                video: toggleStates.video ? '신청 (+SGD 1,000)' : '미신청',
                notes: document.getElementById('notes').value,
                totalPrice: calculateTotal(),
                status: '접수대기',
                timestamp: Date.now()
            };

            try {
                if (isAuthReady) {
                    const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'reservations', newDocId);
                    await setDoc(docRef, newReservation);
                } else {
                    reservations.unshift(newReservation);
                    localStorage.setItem('ahgijagi_sg_reservations', JSON.stringify(reservations));
                }

                copyToClipboard(generateReservationText());
                showToast('🎉 예약이 클라우드 DB로 성공적으로 제출되었습니다!');

                document.getElementById('notes').value = '';

            } catch (err) {
                console.error("Save error:", err);
                showToast('⚠️ 접수 중 오류가 발생했습니다. 다시 시도해 주세요.');
            } finally {
                submitBtn.disabled = false;
                submitBtn.innerHTML = `<i data-lucide="send" class="w-4 h-4"></i><span>예약 신청서 제출하기</span>`;
                lucide.createIcons();
            }
        };

        function renderAdminKPIs() {
            const totalRev = reservations.reduce((sum, r) => sum + (r.status !== '취소' ? Number(r.totalPrice || 0) : 0), 0);
            const totalCount = reservations.length;
            const confirmedCount = reservations.filter(r => r.status === '확정' || r.status === '완료').length;
            const pendingCount = reservations.filter(r => r.status === '접수대기').length;

            const confirmRate = totalCount > 0 ? Math.round((confirmedCount / totalCount) * 100) : 0;

            document.getElementById('kpi-total-revenue').innerText = `SGD ${totalRev.toLocaleString()}`;
            document.getElementById('kpi-total-count').innerText = `${totalCount}건`;
            document.getElementById('kpi-confirm-rate').innerText = `${confirmRate}%`;
            document.getElementById('kpi-pending-count').innerText = `${pendingCount}건`;

            // Badge counter on header
            const badge = document.getElementById('badge-pending-count');
            if (pendingCount > 0) {
                badge.innerText = pendingCount;
                badge.classList.remove('hidden');
            } else {
                badge.classList.add('hidden');
            }

            // Top Package calc
            const counts = { '베이식': 0, '프리미엄': 0, '프레스티지': 0 };
            reservations.forEach(r => { if (counts[r.dolTable] !== undefined) counts[r.dolTable]++; });
            let topPkg = '베이식';
            if (counts['프리미엄'] > counts[topPkg]) topPkg = '프리미엄';
            if (counts['프레스티지'] > counts[topPkg]) topPkg = '프레스티지';
            document.getElementById('kpi-top-package').innerText = topPkg;
        }

        window.copySingleText = function(encodedStr) {
            const decoded = decodeURIComponent(encodedStr);
            copyToClipboard(decoded);
            showToast('해당 예약 내용이 복사되었습니다!');
        };

        window.updateStatus = async function(idKey, newStatus) {
            try {
                if (isAuthReady) {
                    const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'reservations', idKey);
                    await updateDoc(docRef, { status: newStatus });
                } else {
                    const idx = reservations.findIndex(r => (r.dbId || r.id) === idKey);
                    if (idx !== -1) {
                        reservations[idx].status = newStatus;
                        localStorage.setItem('ahgijagi_sg_reservations', JSON.stringify(reservations));
                        refreshDashboardData();
                    }
                }
                showToast('예약 상태가 변경되었습니다.');
            } catch (err) {
                console.error("Status update error:", err);
            }
        };

        window.deleteReservation = async function(idKey) {
            if (confirm('이 예약 내역을 삭제하시겠습니까?')) {
                try {
                    if (isAuthReady) {
                        const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'reservations', idKey);
                        await deleteDoc(docRef);
                    } else {
                        reservations = reservations.filter(r => (r.dbId || r.id) !== idKey);
                        localStorage.setItem('ahgijagi_sg_reservations', JSON.stringify(reservations));
                        refreshDashboardData();
                    }
                    showToast('삭제되었습니다.');
                } catch (err) {
                    console.error("Delete error:", err);
                }
            }
        };

        async function initFirebase() {
            try {
                if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                    await signInWithCustomToken(auth, __initial_auth_token);
                } else {
                    await signInAnonymously(auth);
                }
                isAuthReady = true;

                const colRef = collection(db, 'artifacts', appId, 'public', 'data', 'reservations');
                onSnapshot(colRef, (snapshot) => {
                    reservations = [];
                    snapshot.forEach((docSnap) => {
                        reservations.push({
                            dbId: docSnap.id,
                            ...docSnap.data()
                        });
                    });
                    reservations.sort((a, b) => (b.timestamp || 0) - (a.timestamp || 0));
                    renderCalendar();
                    refreshDashboardData();
                }, (error) => {
                    console.warn("Firestore snapshot error, falling back to local storage:", error);
                    loadFromLocalStorage();
                });
            } catch (err) {
                console.warn("Firebase auth error, falling back to local storage:", err);
                isAuthReady = false;
                loadFromLocalStorage();
            }
        }

        function loadFromLocalStorage() {
            const saved = localStorage.getItem('ahgijagi_sg_reservations');
            if (saved) {
                try {
                    reservations = JSON.parse(saved);
                } catch (e) {
                    reservations = [];
                }
            } else {
                reservations = [];
            }
            renderCalendar();
            refreshDashboardData();
        }
